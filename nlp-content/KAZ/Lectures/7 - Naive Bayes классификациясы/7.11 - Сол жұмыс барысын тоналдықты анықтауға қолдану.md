## Сол жұмыс барысын тоналдыққа қолданамыз

Бұл процесс спамға ғана арналған емес. Мысалдар мен белгілерді ауыстырсақ, сол қадамдар тауар пікірлерін оң немесе теріс деп жіктейді. Төмендегі код барлық 32 пікірді сақтайды: 16 оң және 16 теріс.



```python
positive_reviews = [
    "excellent product works perfectly very happy with this purchase",
    "great quality and fast delivery highly recommend this product",
    "love it exactly as described will definitely buy again",
    "perfect fit and the material feels premium great value",
    "amazing value for money arrived early and works perfectly",
    "very satisfied the battery lasts all day highly recommend",
    "beautiful design and works exactly as advertised very happy",
    "fantastic quality would definitely recommend to others",
    "good product good price and fast delivery very satisfied",
    "works perfectly and the quality is excellent love this",
    "great value for money and it arrived very quickly",
    "happy with this purchase the quality is good",
    "excellent quality fast delivery would buy again",
    "the product works well and feels premium recommend it",
    "very good quality and arrived earlier than expected",
    "perfect product exactly what i wanted very happy",
]

negative_reviews = [
    "terrible quality broke after two days waste of money",
    "very disappointed the item arrived damaged and late",
    "poor build quality and it stopped working immediately",
    "awful experience the product is not worth the price",
    "cheap plastic feels flimsy and broke quickly",
    "worst purchase i have made does not work at all",
    "horrible quality and the delivery was extremely slow",
    "disappointed the battery died within a week",
    "bad quality product broke immediately waste of money",
    "the item arrived damaged and the quality is terrible",
    "poor quality and slow delivery very disappointed",
    "do not buy this it stopped working after a week",
    "terrible product cheap material and bad quality",
    "waste of money the product does not work properly",
    "very poor quality and it arrived late disappointed",
    "awful product broke after two days do not recommend",
]

review_texts = positive_reviews + negative_reviews
review_labels = ["positive"] * len(positive_reviews) + ["negative"] * len(negative_reviews)

r_tr_txt, r_te_txt, r_tr_y, r_te_y = train_test_split(
    review_texts, review_labels, test_size=0.25, random_state=0, stratify=review_labels
)

r_vec = CountVectorizer()
r_clf = MultinomialNB()
r_clf.fit(r_vec.fit_transform(r_tr_txt), r_tr_y)

r_pred = r_clf.predict(r_vec.transform(r_te_txt))

print("accuracy:", accuracy_score(r_te_y, r_pred))
print()
for txt, pred, truth in zip(r_te_txt, r_pred, r_te_y):
    flag = "" if pred == truth else "  <-- wrong"
    print(f"{pred:8} | {txt[:55]}{flag}")
```
