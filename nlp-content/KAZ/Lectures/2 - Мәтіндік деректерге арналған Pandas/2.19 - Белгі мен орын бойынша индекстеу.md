```python
# .loc and .iloc, side by side
print(df.loc[0, "review"])      # by label
print(df.iloc[0, 3])            # by position
print()
print(df.loc[0:2, ["product", "rating"]])   # .loc includes the end, 0,1,2
print()
print(df.iloc[0:2])                          # .iloc excludes it, 0,1
```
