### Underfitting and Overfitting

Недообучение и переобучение

```python
rng = np.random.default_rng(3)
def make_jobs(n):
    x = np.sort(rng.uniform(1, 8, n))
    return x, 10 * x + 2 + rng.normal(0, 6, n)

x_tr, y_tr = make_jobs(8)
x_te, y_te = make_jobs(8)
print("train x:", x_tr.round(1)); print("train y:", y_tr.round(1))
```

```python
xs = np.linspace(0.8, 8.2, 300)
fig, axes = plt.subplots(1, 3, figsize=(11, 3.3), sharey=True)
results = []
for ax, degree in zip(axes, [0, 1, 7]):
    coefs = np.polyfit(x_tr, y_tr, degree)          # least squares fit of a polynomial
    tr_err = mae(np.polyval(coefs, x_tr), y_tr)
    te_err = mae(np.polyval(coefs, x_te), y_te)
    results.append((degree, tr_err, te_err))
    ax.scatter(x_tr, y_tr, label="train", zorder=3)
    ax.scatter(x_te, y_te, marker="x", color="tab:red", label="test", zorder=3)
    ax.plot(xs, np.polyval(coefs, xs), color="black", lw=1)
    ax.set_ylim(-10, 110); ax.set_xlabel("words (hundreds)")
    ax.set_title(f"degree {degree}: train MAE {tr_err:.1f}, test MAE {te_err:.1f}", fontsize=9)
axes[0].set_ylabel("minutes"); axes[0].legend(fontsize=8)
plt.tight_layout()
plt.show()

print(pd.DataFrame(results, columns=["degree", "train MAE", "test MAE"]).round(2).to_string(index=False))
```
