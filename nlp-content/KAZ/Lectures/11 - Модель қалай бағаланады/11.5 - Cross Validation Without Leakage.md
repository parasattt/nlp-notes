### Cross Validation Without Leakage

Кросс-валидация без утечки

```python
from sklearn.pipeline import make_pipeline
from sklearn.model_selection import cross_val_score, KFold

folds = KFold(5, shuffle=True, random_state=0)
cv_rows = []
for a in [0.1, 0.3, 1, 3, 10]:
    pipe = make_pipeline(TfidfVectorizer(min_df=2), Ridge(alpha=a))
    scores = -cross_val_score(pipe, y_train_df["text"], y_train_df["stars"], cv=folds,
                              scoring="neg_mean_absolute_error")
    cv_rows.append((a, scores.mean(), scores.std()))
    print(f"alpha {a:5}: CV MAE per fold {scores.round(3)}  mean {scores.mean():.3f}")

best_alpha = min(cv_rows, key=lambda r: r[1])[0]
print("\nchosen by cross-validation: alpha =", best_alpha)
```
