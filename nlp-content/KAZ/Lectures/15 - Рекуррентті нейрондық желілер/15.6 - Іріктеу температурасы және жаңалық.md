def generate(model, n, temperature=1.0, seed=0):
    g = torch.Generator().manual_seed(seed)
    result = []
    with torch.no_grad():
        for _ in range(n):
            h = torch.zeros(1, model.hidden)
            x, name = 0, ""
            while len(name) < 20:
                h = torch.tanh(model.Wx[x] + h @ model.Wh + model.b)
                p = torch.softmax((h @ model.Wy + model.by) / temperature, dim=-1)
                x = torch.multinomial(p, 1, generator=g).item()
                if x == 0:
                    break
                name += itos[x]
            result.append(name)
    return result
