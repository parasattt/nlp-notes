### Қазақша POS таңбалау

UD Kazakh KTB шағын болғандықтан, ноутбук бастапқы бөліктерді біріктіріп, 900 сөйлемдік қайталанатын оқыту бөлігін жасайды. Қазақ тілінің жалғау-жұрнақтары грамматикалық ақпарат береді; жұрнақ белгілері бар және жоқ модельдерді, әсіресе жаңа сөздерде, салыстырыңыз. Нәтиже шағын бөлініске тәуелді.

```python
kk_counts = defaultdict(Counter)
for s in kk_train:
    for w, t in s:
        kk_counts[w.lower()][t] += 1
kk_best = {w: c.most_common(1)[0][0] for w, c in kk_counts.items()}

kk_gold = np.array([t for s in kk_test for _, t in s])
kk_words = [w.lower() for s in kk_test for w, _ in s]
kk_unknown = np.array([w not in kk_best for w in kk_words])
kk_base = np.array([kk_best.get(w, "NOUN") for w in kk_words])
print(f"share of test words never seen in training: {kk_unknown.mean():.0%}   (English: {unknown.mean():.1%})")
print(f"baseline accuracy: {(kk_base == kk_gold).mean():.3f}")
```

```python
def kk_features(words, i, use_suffixes=True):
    w = words[i]
    lw = w.lower()
    f = {"w=" + lw: 1,
         "prev=" + (words[i - 1].lower() if i > 0 else "<s>"): 1,
         "next=" + (words[i + 1].lower() if i < len(words) - 1 else "</s>"): 1,
         "cap": int(w[0].isupper()), "digit": int(any(ch.isdigit() for ch in w))}
    if use_suffixes:
        for k in (1, 2, 3, 4):
            f[f"suf{k}=" + lw[-k:]] = 1
    return f

def train_kk(use_suffixes):
    X = [kk_features([w for w, _ in s], i, use_suffixes) for s in kk_train for i in range(len(s))]
    y = [t for s in kk_train for _, t in s]
    vec = DictVectorizer()
    model = LogisticRegression(max_iter=1000).fit(vec.fit_transform(X), y)
    X_test = [kk_features([w for w, _ in s], i, use_suffixes) for s in kk_test for i in range(len(s))]
    return model, vec, model.predict(vec.transform(X_test))

for use_suffixes in [False, True]:
    kk_model, kk_vec, kk_pred = train_kk(use_suffixes)
    print(f"suffixes {'on ' if use_suffixes else 'off'}: accuracy {(kk_pred == kk_gold).mean():.3f}   "
          f"on unseen words {(kk_pred[kk_unknown] == kk_gold[kk_unknown]).mean():.3f}")
```

```python
def tag_kk(sentence):
    words = re.findall(r"[\w-]+|[^\w\s]", sentence)
    feats = [kk_features(words, i) for i in range(len(words))]
    return " ".join(f"{w}/{t}" for w, t in zip(words, kk_model.predict(kk_vec.transform(feats))))

print(tag_kk("Біз ертең Алматыға пойызбен барамыз."))
print(tag_kk("Студенттер кітапханада жаңа кітаптарды оқыды."))
```
