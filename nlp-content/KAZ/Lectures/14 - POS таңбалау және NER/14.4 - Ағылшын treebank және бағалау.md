### Ағылшын treebank және бағалау

UD English EWT қолмен белгіленген сөйлемдерді CoNLL-U форматында береді. Түсіндірме, көпсөзді токен және бос түйін жолдарын өткізіңіз; белгілі және белгісіз сөздерді бөлек бағалаңыз. Ноутбуктағы базалық нәтиже шамамен 86%, белгісіз сөздерде 29%.

```python
def read_conllu(path):
    sentences, current = [], []
    for line in open(path, encoding="utf-8"):
        line = line.rstrip("\n")
        if not line:                              # empty line = end of a sentence
            if current:
                sentences.append(current)
                current = []
        elif not line.startswith("#"):
            cols = line.split("\t")
            if "-" in cols[0] or "." in cols[0]:
                continue
            current.append((cols[1], cols[3]))    # (word, tag)
    if current:
        sentences.append(current)
    return sentences

ewt_train = read_conllu("en_ewt-ud-train.conllu")
ewt_test = read_conllu("en_ewt-ud-test.conllu")
print(f"train: {len(ewt_train):,} sentences, {sum(map(len, ewt_train)):,} words")
print(f"test : {len(ewt_test):,} sentences, {sum(map(len, ewt_test)):,} words")
print(ewt_train[5])
```

```python
ewt_counts = defaultdict(Counter)
for s in ewt_train:
    for w, t in s:
        ewt_counts[w.lower()][t] += 1
ewt_best = {w: c.most_common(1)[0][0] for w, c in ewt_counts.items()}

for w in ["book", "back", "like", "that"]:
    print(f"{w:5} {ewt_counts[w].most_common()}")

test_words = [w.lower() for s in ewt_test for w, _ in s]
gold = np.array([t for s in ewt_test for _, t in s])
unknown = np.array([w not in ewt_best for w in test_words])
pred_base = np.array([ewt_best.get(w, "NOUN") for w in test_words])

print()
print(f"baseline accuracy                 : {(pred_base == gold).mean():.3f}")
print(f"share of test words never seen    : {unknown.mean():.1%}")
print(f"accuracy on words seen in training: {(pred_base[~unknown] == gold[~unknown]).mean():.3f}")
print(f"accuracy on unseen words          : {(pred_base[unknown] == gold[unknown]).mean():.3f}")
```
