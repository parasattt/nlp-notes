### Handmade Features

Признаки, созданные вручную

```python
POSITIVE = ["great", "good", "amazing", "love", "best", "delicious", "friendly", "excellent", "awesome", "perfect"]
NEGATIVE = ["bad", "worst", "terrible", "rude", "horrible", "never", "disappointed", "awful", "slow", "not"]

def review_features(text):
    words = re.findall(r"[a-z']+", text.lower())
    return [sum(w in POSITIVE for w in words),
            sum(w in NEGATIVE for w in words),
            len(words) / 100,
            text.count("!")]

F_train = np.array([review_features(t) for t in y_train_df["text"]])
F_test = np.array([review_features(t) for t in y_test_df["text"]])
print("F_train shape:", F_train.shape)

four = LinearRegression().fit(F_train, y_train_df["stars"])
for name, w in zip(["positive words", "negative words", "length (100 words)", "exclamation marks"], four.coef_):
    print(f"weight of {name:20}: {w:+.3f}")
print(f"bias                          : {four.intercept_:+.3f}")
print()
print(f"test MAE: {mae(four.predict(F_test), y_test_df['stars']):.3f}  (baseline 1.173)")
```
