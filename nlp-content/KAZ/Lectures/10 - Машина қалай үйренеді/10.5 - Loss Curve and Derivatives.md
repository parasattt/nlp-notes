### Loss Curve and Derivatives

Кривая потерь и производная

```python
ws = np.linspace(5, 15, 200)
losses = [mse(predict(x, w, 1.8), y) for w in ws]
best_w = ws[np.argmin(losses)]

plt.figure(figsize=(5, 3.2))
plt.plot(ws, losses)
plt.scatter([best_w], [min(losses)], color="red", zorder=3, label=f"lowest at w = {best_w:.2f}")
plt.xlabel("w (minutes per 100 words)"); plt.ylabel("MSE")
plt.title("Loss for each value of w (b fixed at 1.8)", fontsize=10)
plt.legend(fontsize=8)
plt.tight_layout()
plt.show()
```

```python
def loss_w(w, b=1.8):
    return mse(predict(x, w, b), y)

w0, nudge = 7.0, 0.001
slope = (loss_w(w0 + nudge) - loss_w(w0)) / nudge
print(f"loss at w = {w0}      : {loss_w(w0):.4f}")
print(f"loss at w = {w0 + nudge}  : {loss_w(w0 + nudge):.4f}")
print(f"slope = change / nudge: {slope:.2f}")
```

```python
def slopes(w, b):
    error = predict(x, w, b) - y
    return 2 * np.mean(error * x), 2 * np.mean(error)

print("formula slope for w at w = 7:", round(slopes(7.0, 1.8)[0], 2))
```
