### Names data n-grams and loss

The notebook holds out 3,000 names for testing. Average negative log likelihood measures how much probability the model assigns to the actual next letters, including the end marker; lower is better. Longer n-grams capture context but sparse tables grow quickly.

```python
url = "https://raw.githubusercontent.com/karpathy/makemore/master/names.txt"
names = urllib.request.urlopen(url).read().decode("utf-8").split()
print(len(names), "names, for example", names[:8])

random.Random(0).shuffle(names)
test_names, train_names = names[:3000], names[3000:]
print(f"train {len(train_names):,}   test {len(test_names):,}")
print("longest name:", max(names, key=len), max(map(len, names)), "letters")
```

```python
def train_ngram(names, n):
    counts = defaultdict(Counter)
    for name in names:
        s = "." * (n - 1) + name + "."          # n - 1 dots, so the first letter also has a context
        for i in range(n - 1, len(s)):
            counts[s[i - n + 1:i]][s[i]] += 1
    return counts

def ngram_prob(counts, context, letter, k=0.1):
    c = counts.get(context, {})
    return (c.get(letter, 0) + k) / (sum(c.values()) + k * 27)

def ngram_loss(counts, n, names):
    total, count, rare = 0.0, 0, 0
    for name in names:
        s = "." * (n - 1) + name + "."
        for i in range(n - 1, len(s)):
            context = s[i - n + 1:i]
            total -= math.log(ngram_prob(counts, context, s[i]))
            count += 1
            rare += sum(counts.get(context, {}).values()) < 5     # context seen fewer than 5 times
    return total / count, rare / count

rows = []
for n in [1, 2, 3, 4, 5]:
    counts = train_ngram(train_names, n)
    loss, rare = ngram_loss(counts, n, test_names)
    rows.append({"letters of memory": n - 1, "test loss": round(loss, 3),
                 "perplexity": round(math.exp(loss), 1), "contexts in the table": len(counts),
                 "test letters with a rare context": f"{rare:.1%}"})
ngram_results = pd.DataFrame(rows)
ngram_results
```

def generate_ngram(counts, n, rng):
    name = "." * (n - 1)
    while len(name) < 30:
        c = counts.get(name[len(name) - n + 1:] if n > 1 else "", {})
        letter = rng.choices(list(c), weights=list(c.values()))[0]
        if letter == ".":
            break
        name += letter
    return name.strip(".")

for n in [2, 4]:
    counts = train_ngram(train_names, n)
    rng = random.Random(0)
    print(f"{n - 1} letter(s) of memory:", [generate_ngram(counts, n, rng) for _ in range(10)])
