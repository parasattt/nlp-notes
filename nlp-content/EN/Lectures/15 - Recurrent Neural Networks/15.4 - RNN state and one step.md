### RNN state and one step

The hidden vector is a fixed-size memory updated at every character: h_new = tanh(xW_x + h_old W_h + b). Output scores become probabilities through softmax. Different histories can produce different predictions even when the last letter is the same.

```python
for z in [-3, -1, 0, 0.5, 1, 3]:
    print(f"tanh({z:4}) = {math.tanh(z):6.3f}")
```

```python
alphabet = [".", "a", "n"]
Wx = np.array([[0.0, 0.0],      # row for "."
               [1.0, 0.0],      # row for "a"
               [0.0, 1.0]])     # row for "n"
Wh = np.array([[0.5, 0.5],
               [-0.5, 0.5]])
Wy = np.array([[1.0, 2.0, -1.0],
               [0.0, -1.0, 2.0]])   # 2 neurons -> 3 scores (for ".", "a", "n")

def softmax(scores):
    e = np.exp(scores - scores.max())
    return e / e.sum()

def rnn_step(h, letter):
    return np.tanh(Wx[alphabet.index(letter)] + h @ Wh)
```

```python
for text in ["a", "na"]:
    h = np.zeros(2)
    for letter in text:
        h = rnn_step(h, letter)
    p = softmax(h @ Wy)
    print(f"read {text!r:5} h = {np.round(h, 3)}   P(next) = " +
          ", ".join(f"{l!r}: {q:.2f}" for l, q in zip(alphabet, p)))
```

```python
name = ".an."
h, losses = np.zeros(2), []
for cur, nxt in zip(name, name[1:]):
    h = rnn_step(h, cur)
    p = softmax(h @ Wy)
    losses.append(-math.log(p[alphabet.index(nxt)]))
    print(f"read {cur!r}  P({nxt!r}) = {p[alphabet.index(nxt)]:.3f}   -log = {losses[-1]:.3f}")
print(f"loss of this name = average = {np.mean(losses):.3f}")
```
