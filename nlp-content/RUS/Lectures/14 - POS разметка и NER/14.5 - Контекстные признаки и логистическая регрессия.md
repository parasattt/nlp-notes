### Контекстные признаки и логистическая регрессия

Используйте слово, соседей, приставки, суффиксы, заглавные буквы и цифры как признаки. DictVectorizer строит матрицу, логистическая регрессия предсказывает POS. В ноутбуке результат около 94% в целом и 80% на неизвестных словах; анализируйте ошибки.

```python
def word_features(words, i):
    w = words[i]
    lw = w.lower()
    return {
        "w=" + lw: 1,
        "prev=" + (words[i - 1].lower() if i > 0 else "<s>"): 1,
        "next=" + (words[i + 1].lower() if i < len(words) - 1 else "</s>"): 1,
        "suf2=" + lw[-2:]: 1,
        "suf3=" + lw[-3:]: 1,
        "pre2=" + lw[:2]: 1,
        "cap": int(w[0].isupper()),
        "digit": int(any(ch.isdigit() for ch in w)),
        "hyph": int("-" in w),
        "first": int(i == 0),
    }

print(word_features(["we", "book", "a", "table"], 1))
```

```python
def sentences_to_xy(sentences):
    X, y = [], []
    for sent in sentences:
        words = [w for w, _ in sent]
        for i, (_, tag) in enumerate(sent):
            X.append(word_features(words, i))
            y.append(tag)
    return X, y
```

```python
def tag(sentence):
    words = re.findall(r"\w+|[^\w\s]", sentence)     # words and punctuation marks
    feats = [word_features(words, i) for i in range(len(words))]
    return list(zip(words, tagger.predict(vectorizer.transform(feats))))

for s in ["We want to book a table for two.", "I read the book in one evening.",
          "The waiter was blorfingly slow."]:
    print(" ".join(f"{w}/{t}" for w, t in tag(s)))
```
