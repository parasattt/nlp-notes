### Частотная базовая модель

Назначайте слову самую частую метку в обучении, а неизвестному — NOUN. Базовая модель игнорирует контекст и ошибается на неоднозначных и новых словах.

```python
tagged_toy = [
    [("I", "PRON"), ("read", "VERB"), ("a", "DET"), ("good", "ADJ"), ("book", "NOUN")],
    [("the", "DET"), ("book", "NOUN"), ("was", "AUX"), ("long", "ADJ")],
    [("she", "PRON"), ("bought", "VERB"), ("the", "DET"), ("book", "NOUN")],
    [("we", "PRON"), ("book", "VERB"), ("a", "DET"), ("table", "NOUN")],
    [("the", "DET"), ("table", "NOUN"), ("was", "AUX"), ("free", "ADJ")],
    [("I", "PRON"), ("will", "AUX"), ("book", "VERB"), ("a", "DET"), ("room", "NOUN")],
]

tag_counts = defaultdict(Counter)
for sent in tagged_toy:
    for word, tag in sent:
        tag_counts[word.lower()][tag] += 1

for w in ["book", "the", "table", "was"]:
    print(f"{w:6} {dict(tag_counts[w])}")
```

```python
most_frequent = {w: c.most_common(1)[0][0] for w, c in tag_counts.items()}

def baseline_tag(words):
    return [most_frequent.get(w.lower(), "NOUN") for w in words]

for sentence in ["I read the book", "we book a room", "they book a flight"]:
    words = sentence.split()
    print(list(zip(words, baseline_tag(words))))
```
