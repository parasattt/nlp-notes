### Данные CoNLL и обученная NER

CoNLL-2003 содержит тексты Reuters с BIO-разметкой. Предсказание O для всех токенов даёт высокую точность по токенам, но нулевой F1 сущностей. Словарь с самым длинным совпадением сравнивается с классификатором по форме слова и контексту: в ноутбуке F1 около 0.59 и 0.78.

```python
base = "https://huggingface.co/api/datasets/tner/conll2003/parquet/conll2003/"
conll_train = pd.read_parquet(base + "train/0.parquet")
conll_test = pd.read_parquet(base + "test/0.parquet")

label_names = {0: "O", 1: "B-ORG", 2: "B-MISC", 3: "B-PER", 4: "I-PER",
               5: "B-LOC", 6: "I-ORG", 7: "I-MISC", 8: "I-LOC"}

def to_sentences(df):
    return [(list(words), [label_names[t] for t in tags]) for words, tags in zip(df["tokens"], df["tags"])]

train_ner = to_sentences(conll_train)
test_ner = to_sentences(conll_test)
test_true = [tags for _, tags in test_ner]
print(f"train: {len(train_ner):,} sentences   test: {len(test_ner):,} sentences")
for words, tags in train_ner[:2]:
    print(list(zip(words, tags)))
```

```python
train_names = {}
for words, tags in train_ner:
    for start, end, kind in bio_to_spans(tags):
        train_names[tuple(words[start:end])] = kind
print(f"names in the list: {len(train_names):,}")

def list_tagger(words):
    tags, i = ["O"] * len(words), 0
    while i < len(words):
        for n in (4, 3, 2, 1):                           # try the longest name first
            name = tuple(words[i:i + n])
            if i + n <= len(words) and name in train_names:
                kind = train_names[name]
                tags[i] = "B-" + kind
                for j in range(i + 1, i + n):
                    tags[j] = "I-" + kind
                i += n
                break
        else:
            i += 1
    return tags

list_pred = [list_tagger(words) for words, _ in test_ner]
print("list baseline: precision, recall, F1 =", entity_scores(test_true, list_pred))
```

```python
def shape(w):
    s = re.sub("[A-Z]", "X", w)
    s = re.sub("[a-z]", "x", s)
    s = re.sub("[0-9]", "d", s)
    return re.sub(r"(.)\1+", r"\1\1", s)       # "Xxxxxx" -> "Xxx": a run of 2 or more becomes 2

print([shape(w) for w in ["Almaty", "KASPI", "iPhone", "1996-08-22"]])

def ner_features(words, i):
    w = words[i]
    lw = w.lower()
    f = {"w=" + lw: 1, "shape=" + shape(w): 1, "suf3=" + lw[-3:]: 1, "pre3=" + lw[:3]: 1,
         "cap": int(w[:1].isupper()), "allcaps": int(w.isupper()), "first": int(i == 0)}
    for d in (-2, -1, 1, 2):                    # the two words on each side
        j = i + d
        other = words[j] if 0 <= j < len(words) else "<pad>"
        f[f"shape{d:+d}=" + shape(other)] = 1
        f[f"w{d:+d}=" + other.lower()] = 1
    return f

print(ner_features("Aigerim Bekova joined Kaspi Bank".split(), 3))
```

```python
X_train = [ner_features(words, i) for words, _ in train_ner for i in range(len(words))]
y_train = [t for _, tags in train_ner for t in tags]
X_test = [ner_features(words, i) for words, _ in test_ner for i in range(len(words))]

ner_vec = DictVectorizer()
ner_model = LogisticRegression(max_iter=300)
ner_model.fit(ner_vec.fit_transform(X_train), y_train)
flat = list(ner_model.predict(ner_vec.transform(X_test)))

lr_pred, k = [], 0                                  # cut the flat list back into sentences
for words, _ in test_ner:
    lr_pred.append(flat[k:k + len(words)])
    k += len(words)

print("word accuracy:", round(np.mean([a == b for t, p in zip(test_true, lr_pred) for a, b in zip(t, p)]), 3))
print("logistic regression: precision, recall, F1 =", entity_scores(test_true, lr_pred))
print()
for kind in ["PER", "ORG", "LOC", "MISC"]:
    only = lambda s: [t if t[2:] == kind else "O" for t in s]
    print(f"  {kind:4} P, R, F1 = {entity_scores([only(t) for t in test_true], [only(p) for p in lr_pred])}")
```

```python
def ner(sentence):
    words = re.findall(r"\w+(?:[-'.]\w+)*|[^\w\s]", sentence)
    feats = [ner_features(words, i) for i in range(len(words))]
    tags = ner_model.predict(ner_vec.transform(feats))
    return [(" ".join(words[s:e]), kind) for s, e, kind in bio_to_spans(tags)]

for s in ["Michael Jordan visited Jordan .",
          "Denzel Washington bought an Apple Watch .",
          "Freedom Finance opened an office in Shymkent .",
          "Aigerim Bekova joined Kaspi Bank in Almaty after five years at Halyk ."]:
    print(f"{s:72} {ner(s)}")
```
