```python
class CharRNN(nn.Module):
    def __init__(self, n_symbols=27, hidden=128):
        super().__init__()
        self.hidden = hidden
        self.Wx = nn.Parameter(torch.randn(n_symbols, hidden) * 0.1)
        self.Wh = nn.Parameter(torch.randn(hidden, hidden) * hidden ** -0.5)
        self.b = nn.Parameter(torch.zeros(hidden))
        self.Wy = nn.Parameter(torch.randn(hidden, n_symbols) * 0.1)
        self.by = nn.Parameter(torch.zeros(n_symbols))

    def forward(self, X):
        h = torch.zeros(X.shape[0], self.hidden)        # empty memory for every name
        scores = []
        for t in range(X.shape[1]):                     # one letter at a time
            h = torch.tanh(self.Wx[X[:, t]] + h @ self.Wh + self.b)
            scores.append(h @ self.Wy + self.by)
        return torch.stack(scores, dim=1)              # names x steps x 27 scores

torch.manual_seed(0)
model = CharRNN()
print("numbers to learn:", sum(p.numel() for p in model.parameters()))
```

```python
loss_fn = nn.CrossEntropyLoss(ignore_index=-1)

def loss_on(model, X, Y):
    with torch.no_grad():                               # no gradients needed, only the number
        return loss_fn(model(X).reshape(-1, 27), Y.reshape(-1)).item()

print("test loss before training:", round(loss_on(model, X_test, Y_test), 3))
```

```python
def train(model, X, Y, epochs, lr, batch_size=64, X_check=None, Y_check=None):
    optimizer = torch.optim.Adam(model.parameters(), lr=lr)
    history = []
    for epoch in range(epochs):
        if epoch == int(epochs * 0.7):                  # smaller steps for the last 30% of epochs
            for group in optimizer.param_groups:
                group["lr"] = lr / 10
        order = torch.randperm(len(X))
        for start in range(0, len(X), batch_size):
            batch = order[start:start + batch_size]
            loss = loss_fn(model(X[batch]).reshape(-1, 27), Y[batch].reshape(-1))
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
        if X_check is not None:
            history.append((loss_on(model, X, Y), loss_on(model, X_check, Y_check)))
    return history

torch.manual_seed(0)
model = CharRNN()
history = train(model, X_train, Y_train, epochs=20, lr=0.005, X_check=X_test, Y_check=Y_test)
for epoch in [0, 4, 9, 14, 19]:
    print(f"epoch {epoch + 1:2}   train loss {history[epoch][0]:.3f}   test loss {history[epoch][1]:.3f}")
```
