### Биграммы букв вручную

Добавьте маркеры начала и конца и посчитайте соседние пары. Вероятность следующей буквы равна частоте пары, делённой на число продолжений текущей буквы. Биграмма помнит только одну букву, может копировать имена, рано останавливаться или повторяться.

```python
toy = ["dana", "dias", "aida", "asan", "sana", "nadia"]
```

```python
pairs = defaultdict(Counter)
for name in toy:
    s = "." + name + "."
    for cur, nxt in zip(s, s[1:]):
        pairs[cur][nxt] += 1

table = pd.DataFrame(pairs).T.fillna(0).astype(int)
table = table.loc[[".", "a", "d", "i", "n", "s"], [".", "a", "d", "i", "n", "s"]]
table["total"] = table.sum(axis=1)
table
```

```python
def generate_bigram(pairs, rng):
    name, cur = "", "."
    while True:
        options = pairs[cur]
        cur = rng.choices(list(options), weights=list(options.values()))[0]
        if cur == ".":
            return name
        name += cur

rng = random.Random(1)
print([generate_bigram(pairs, rng) for _ in range(12)])
```
